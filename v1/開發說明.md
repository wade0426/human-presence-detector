# 人體辨識休息提醒系統 開發文檔

## 專案概述

本專案是一個基於電腦視覺的健康提醒應用程式，透過攝像頭即時監控使用者是否在電腦前工作。當系統檢測到使用者持續工作 1 小時時，會提醒使用者該休息了。若在計時期間使用者離開超過 5 分鐘，系統將自動重置計時器。

## 技術架構

### 核心技術棧
- **電腦視覺**: YOLO (You Only Look Once) v8 人體檢測模型
- **影像處理**: OpenCV
- **GUI框架**: Tkinter
- **程式語言**: Python 3.8+

### 系統架構
```
攝像頭輸入 → YOLO模型檢測 → 狀態管理 → 計時邏輯 → UI提醒
```

## 環境需求與安裝

### 系統需求
- Windows 10/11
- Python 3.8 或以上版本
- 可用的攝像頭設備

## 核心功能實作

### 1. 主程式架構 (main.py)
```
import cv2
import time
import threading
from datetime import datetime, timedelta
from ultralytics import YOLO
import tkinter as tk
from tkinter import messagebox, ttk

class PersonDetectionSystem:
    def __init__(self):
        self.model = YOLO(r'D:\Code\Models\yolov8n.pt')
        self.cap = cv2.VideoCapture(0)
        self.is_running = False
        
        # 狀態變數
        self.person_detected = False
        self.work_start_time = None
        self.away_start_time = None
        
        # 時間設定 (秒)
        self.WORK_DURATION = 3600  # 1小時 = 3600秒
        self.AWAY_THRESHOLD = 300  # 5分鐘 = 300秒
        
        self.setup_ui()
    
    def setup_ui(self):
        """建立使用者介面"""
        self.root = tk.Tk()
        self.root.title("休息提醒系統")
        self.root.geometry("400x300")
        
        # 狀態顯示
        self.status_label = tk.Label(
            self.root, 
            text="系統待機中...", 
            font=("Arial", 14)
        )
        self.status_label.pack(pady=20)
        
        # 計時顯示
        self.time_label = tk.Label(
            self.root, 
            text="工作時間: 00:00:00", 
            font=("Arial", 12)
        )
        self.time_label.pack(pady=10)
        
        # 控制按鈕
        self.start_btn = tk.Button(
            self.root, 
            text="開始監控", 
            command=self.start_monitoring,
            font=("Arial", 12)
        )
        self.start_btn.pack(pady=10)
        
        self.stop_btn = tk.Button(
            self.root, 
            text="停止監控", 
            command=self.stop_monitoring,
            font=("Arial", 12)
        )
        self.stop_btn.pack(pady=5)
    
    def detect_person(self, frame):
        """人體檢測函數"""
        results = self.model(frame, classes=, verbose=False)
        return len(results.boxes) > 0
    
    def monitoring_loop(self):
        """主要監控循環"""
        while self.is_running:
            ret, frame = self.cap.read()
            if not ret:
                continue
            
            # 檢測人體
            current_person_detected = self.detect_person(frame)
            
            # 狀態轉換邏輯
            self.handle_detection_state(current_person_detected)
            
            # 更新UI
            self.update_ui()
            
            time.sleep(1)  # 每秒檢測一次
    
    def handle_detection_state(self, current_detected):
        """處理檢測狀態變化"""
        now = datetime.now()
        
        if current_detected:
            # 檢測到人
            if not self.person_detected:
                # 人剛回來
                self.person_detected = True
                if self.work_start_time is None:
                    # 開始新的工作計時
                    self.work_start_time = now
                self.away_start_time = None
        else:
            # 沒檢測到人
            if self.person_detected:
                # 人剛離開
                self.person_detected = False
                self.away_start_time = now
        
        # 檢查離開時間是否超過閾值
        if (not self.person_detected and 
            self.away_start_time and 
            (now - self.away_start_time).total_seconds() > self.AWAY_THRESHOLD):
            # 重置工作計時
            self.reset_work_timer()
        
        # 檢查工作時間是否達到1小時
        if (self.person_detected and 
            self.work_start_time and 
            (now - self.work_start_time).total_seconds() >= self.WORK_DURATION):
            self.show_rest_reminder()
    
    def reset_work_timer(self):
        """重置工作計時器"""
        self.work_start_time = None
        self.away_start_time = None
    
    def show_rest_reminder(self):
        """顯示休息提醒"""
        messagebox.showwarning(
            "休息提醒", 
            "您已經持續工作1小時了！\n該起來休息一下囉～"
        )
        self.reset_work_timer()
    
    def update_ui(self):
        """更新使用者介面"""
        # 更新狀態
        if self.person_detected:
            status_text = "✅ 檢測到使用者"
        else:
            status_text = "❌ 使用者不在"
        
        # 更新工作時間
        if self.work_start_time:
            elapsed = datetime.now() - self.work_start_time
            time_str = str(elapsed).split('.')  # 移除微秒
        else:
            time_str = "00:00:00"
        
        # 在主執行緒更新UI
        self.root.after(0, lambda: self.status_label.config(text=status_text))
        self.root.after(0, lambda: self.time_label.config(text=f"工作時間: {time_str}"))
    
    def start_monitoring(self):
        """開始監控"""
        if not self.is_running:
            self.is_running = True
            self.monitoring_thread = threading.Thread(target=self.monitoring_loop)
            self.monitoring_thread.daemon = True
            self.monitoring_thread.start()
            self.start_btn.config(state="disabled")
            self.stop_btn.config(state="normal")
    
    def stop_monitoring(self):
        """停止監控"""
        self.is_running = False
        self.reset_work_timer()
        self.start_btn.config(state="normal")
        self.stop_btn.config(state="disabled")
    
    def run(self):
        """執行應用程式"""
        self.root.protocol("WM_DELETE_WINDOW", self.on_closing)
        self.root.mainloop()
    
    def on_closing(self):
        """關閉應用程式"""
        self.is_running = False
        if self.cap:
            self.cap.release()
        cv2.destroyAllWindows()
        self.root.destroy()

if __name__ == "__main__":
    app = PersonDetectionSystem()
    app.run()
```

## 使用說明

### 執行步驟
1. 確保攝像頭正常運作
2. 執行主程式: `python main.py`
3. 點擊「開始監控」按鈕
4. 系統開始監控並計時
5. 達到1小時會自動彈出休息提醒

### 功能說明
- **自動檢測**: 每秒檢測一次是否有人在攝像頭前
- **智能計時**: 只在檢測到人時計算工作時間
- **自動重置**: 離開超過5分鐘自動重置計時器
- **友善提醒**: 達到1小時彈出提醒視窗

```
import json
from datetime import datetime

class WorkLogger:
    def __init__(self, log_file="work_log.json"):
        self.log_file = log_file
    
    def log_work_session(self, start_time, end_time):
        """記錄工作時段"""
        session = {
            "start": start_time.isoformat(),
            "end": end_time.isoformat(),
            "duration": (end_time - start_time).total_seconds()
        }
        
        try:
            with open(self.log_file, 'r') as f:
                logs = json.load(f)
        except FileNotFoundError:
            logs = []
        
        logs.append(session)
        
        with open(self.log_file, 'w') as f:
            json.dump(logs, f, indent=2)
```

### 2. 設定檔管理
```
import configparser

class ConfigManager:
    def __init__(self, config_file="settings.ini"):
        self.config_file = config_file
        self.config = configparser.ConfigParser()
        self.load_config()
    
    def load_config(self):
        """載入設定檔"""
        self.config.read(self.config_file)
        if 'DEFAULT' not in self.config:
            self.create_default_config()
    
    def create_default_config(self):
        """建立預設設定"""
        self.config['DEFAULT'] = {
            'work_duration': '3600',
            'away_threshold': '300',
            'camera_index': '0'
        }
        self.save_config()
    
    def save_config(self):
        """儲存設定檔"""
        with open(self.config_file, 'w') as f:
            self.config.write(f)
```

```
def preprocess_frame(frame):
    """預處理影像以提升檢測效能"""
    # 縮放影像
    height, width = frame.shape[:2]
    if width > 640:
        scale = 640 / width
        new_width = int(width * scale)
        new_height = int(height * scale)
        frame = cv2.resize(frame, (new_width, new_height))
    
    return frame
```
